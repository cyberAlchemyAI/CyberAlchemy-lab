    def test_large_json_dependency_exact_roundtrip(self):
        raw = T.encoded({'public_source_snapshot': 'x' * (T.MAX_BYTES + 1)})
        self.change_dependency(raw, 'json-utf8')
        self.contract['maximum_package_bytes'] = T.MAX_PACKAGE_BYTES
        self.sync_export_controls()
        archive = self.export()
        self.assertGreater(len(archive), T.MAX_BYTES)
        package = self.engine.verify_archive(archive, T.pin('package:large-json', archive))
        item = next(r for r in package['materials'] if r['binding'] == self.dependency_binding)
        self.assertEqual(base64.b64decode(item['original_bytes_b64']), raw)
        with self.assertRaises(ValueError):
            T.parse(archive)
