operation AskQuestion:
  input current optional<Session>
  input invocation Invocation
  output optional<Session>
  when not (match invocation Command(value) => true else false):
    return current
  when not (match CommandOf(invocation).payload Ask(value) => true else false):
    return current
  on absent(current) reject session-missing
  on not ContextSufficient(current) reject context-insufficient

  input session Session
  input change Payload
  input actor Ref

  field responses <- match change Respond(value) => append(session.responses,Response(response_id=value.response_id,question_id=value.question_id,content=value.content,status=value.status,actor=actor,evidence=value.evidence)) else session.responses

module Define version 1
use Define.Shared.Digest
use Define.Packs.PrepareCandidate.PrepareCandidate
use Define.Packs.PrepareRequest
use Define.Shared.Ref
use Define.Work.Session
use Define.Shared.Standing
use Define.Shared.AuthorityEffect

entity PreparationRun:
  id UInt64
  project Ref
  work Ref
  source_digest Digest
  session_snapshot Session
  identity id

entity PreparationDecision:
  id UInt64
  stageRevision UInt64
  kind literal "complete"
  identity id, stageRevision

operation BeginPreparation:
  input id UInt64
  input session Session
  input source_digest Digest
  output PreparationRun
  requires id > 0 on-false reject invalid-run-id
  transaction BeginPreparation:
    atomic-participants PreparationRun
    create run PreparationRun:
      id <- id
      project <- session.scope.project
      work <- session.scope.work
      source_digest <- source_digest
      session_snapshot <- session
    on-unique-conflict reject duplicate-preparation
    on-proven-rollback fail preparation-rolled-back
    on-unknown-commit fail preparation-outcome-unknown
  return run

operation ContinuePreparation:
  input id UInt64
  input stageResult PreparedCandidate
  input stageRevision UInt64
  output PreparationDecision
  requires stageResult.id == id on-false reject preparation-mismatch
  read run from Define.ReadPreparationRunEntry(id=id)
    on-missing reject preparation-missing
    on-unavailable fail preparation-unavailable
  read stored from Define.ReadPreparedCandidateEntry(id=id,attempt=stageResult.attempt)
    on-missing reject candidate-missing
    on-unavailable fail candidate-unavailable
  on stageResult != stored reject candidate-result-mismatch
  on stored.source_digest != run.source_digest reject source-mismatch
  on stored.request.request_id != run.session_snapshot.session_id reject request-mismatch
  on stored.request.request_revision != run.session_snapshot.request_revision reject request-revision-mismatch
  transaction ContinuePreparation:
    atomic-participants PreparationDecision
    create decision PreparationDecision:
      id <- id
      stageRevision <- stageRevision
      kind <- "complete"
    on-unique-conflict reject duplicate-preparation
    on-proven-rollback fail preparation-rolled-back
    on-unknown-commit fail preparation-outcome-unknown
  return decision

    authenticated customer
    permission define.BeginPreparationEntry

  attempt UInt64
  request_id Ref
  request_revision UInt64
  project Ref
  work Ref
  source_digest Digest
  status Standing
  authority_effect AuthorityEffect

event CandidateAttemptPrepared version 1:
  payload PreparedNotice
  from Define.PublishPreparation
  delivered-to LabIntegration.ReceiveCandidateEntry


operation PublishPreparation:
  input id UInt64
  input session Session
  input source_digest Digest
  input stageAttempt UInt64
  output PreparedCandidate
  on stageAttempt == 0 reject invalid-stage-attempt
  read run from Define.ReadPreparationRunEntry(id=id)
    on-missing reject preparation-missing
    on-unavailable fail preparation-unavailable
  on session != run.session_snapshot reject preparation-session-mismatch
  on source_digest != run.source_digest reject preparation-source-mismatch
  call request Define.Packs.PrepareCandidate.PrepareCandidate(session=session)
  transaction PrepareCandidate:
    atomic-participants PreparedCandidate, CandidateAttemptPrepared
    create prepared PreparedCandidate:
      id <- id
      attempt <- stageAttempt
      request <- request
      source_digest <- source_digest
    emit CandidateAttemptPrepared payload PreparedNotice(id=id,attempt=stageAttempt,request_id=request.request_id,request_revision=request.request_revision,project=session.scope.project,work=session.scope.work,source_digest=source_digest,status=Candidate,authority_effect=None)
    on-unique-conflict reject duplicate-preparation
    on-proven-rollback fail preparation-rolled-back
    on-unknown-commit fail preparation-outcome-unknown
  return prepared
